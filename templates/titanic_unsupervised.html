Clustering & Reduction Codebase

K-Means partitions observations using squared Euclidean distances. PCA converts correlated columns into linearly uncorrelated dimensions.

# Apply PCA to reduce dimensions for plotting from sklearn.decomposition import PCA from sklearn.preprocessing import MinMaxScaler import pandas as pd # Load and preprocess original features X = df[['Pclass', 'Sex', 'Age', 'SibSp', 'Parch', 'Fare', 'Embarked', 'FamilySize', 'Title', 'AgeCategory']] X_scaled = MinMaxScaler().fit_transform(X) # Reduce features to 2 principal components pca = PCA(n_components=2, random_state=42) X_pca = pca.fit_transform(X_scaled) # View explained variance metrics print(f"Explained Variance Ratio: {pca.explained_variance_ratio_}")
# Apply K-Means Clustering to group data structures from sklearn.cluster import KMeans # Set cluster groupings K n_clusters = 3 # Fit K-Means on scaled original features kmeans = KMeans(n_clusters=n_clusters, random_state=42, n_init=10) kmeans.fit(X_scaled) # Assign cluster classifications labels = kmeans.labels_ print(f"Data clustered successfully with K={n_clusters}.")
